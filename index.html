        /**

         * Formats numbers up to 2 decimal places without trailing zeros.

         */

        function fmtNum(val) {

            return Number.isInteger(val)

                ? val.toLocaleString('en-US')

                : Number(val.toFixed(2)).toLocaleString('en-US');

        }


        /**

         * Renders the 1-9 segmented contributor selector buttons.

         */

        function renderButtons() {

            const container = document.getElementById('number-buttons');

            if (!container) return;

            container.innerHTML = '';


            for (let i = 1; i <= 9; i++) {

                const btn = document.createElement('button');

                btn.type = 'button';

                btn.className = 'num-btn' + (i === selectedCount ? ' active' : '');

                btn.textContent = i;

                btn.setAttribute('aria-pressed', i === selectedCount ? 'true' : 'false');

                btn.onclick = () => selectCount(i);

                container.appendChild(btn);

            }

        }


        /**

         * Updates the selected contributor count (1-9) and re-renders the table.

         */

        function selectCount(count) {

            selectedCount = count;

            renderButtons();

            renderTable();

        }

        /**

         * Renders the responsibility table rows ending at Kid 9.

         */

        function renderTable() {

            const tbody = document.getElementById('table-body');

            if (!tbody) return;

            tbody.innerHTML = '';


            const t = i18n[currentLang];

            const isAr = currentLang === 'ar';

            const cellDir = isAr ? 'rtl' : 'ltr';

            const cellClass = isAr ? 'cell-rtl' : '';


            // Exact count of rows ending with 9

            const startKid = 10 - selectedCount;


            // Equal division of 642 hours and 24-hour day equivalent

            const hoursEach = TOTAL_HOURS / selectedCount;

            const daysEach = hoursEach / 24;

            const hoursDollarEach = hoursEach * HOURLY_RATE;


            const hoursFormatted = fmtNum(hoursEach);

            const daysFormatted = fmtNum(daysEach);

            const hoursDollarFormatted = fmtNum(hoursDollarEach);


            for (let kid = startKid; kid <= 9; kid++) {

                const tr = document.createElement('tr');

                tr.style.animationDelay = `${(kid - startKid) * 25}ms`;


                // Expense percentage split: Kid 9 takes 100% if alone, else 75% (25% split among rest)

                let expensePct = 0;

                if (selectedCount === 1) {

                    expensePct = 100;

                } else if (kid === 9) {

                    expensePct = 75;

                } else {

                    expensePct = 25 / (selectedCount - 1);

                }


                const expenseDollar = (expensePct / 100) * SHARED_EXPENSES;

                const totalDollar = hoursDollarEach + expenseDollar;


                const pctFormatted = fmtNum(expensePct);

                const expenseDollarFormatted = fmtNum(expenseDollar);

                const totalDollarFormatted = fmtNum(totalDollar);


                // Column 1: Kids

                const tdKid = document.createElement('td');

                tdKid.innerHTML = `

                    <div class="kid-pill-wrap">

                        <span class="kid-badge">${kid}</span>

                    </div>

                `;

                tr.appendChild(tdKid);


                // Column 2: Share of Hours

                const tdHours = document.createElement('td');

                tdHours.dir = cellDir;

                tdHours.className = cellClass;

                tdHours.innerHTML = `

                    <span class="val-main"><span class="bidi-num">${hoursFormatted}</span> ${t.hrs}</span>

                    <span class="muted-paren">(<span class="bidi-num">${daysFormatted}</span> ${t.days})</span>

                    <span class="word-or">${t.or}</span>

                    <span class="val-main"><span class="bidi-num">$${hoursDollarFormatted}</span></span>

                `;

                tr.appendChild(tdHours);


                // Column 3: Share of Expenses

                const tdExpenses = document.createElement('td');

                tdExpenses.dir = cellDir;

                tdExpenses.className = cellClass;

                tdExpenses.innerHTML = `

                    <span class="val-main"><span class="bidi-num">$${expenseDollarFormatted}</span></span>

                    <span class="muted-paren">(<span class="bidi-num">${pctFormatted}%</span>)</span>

                `;

                tr.appendChild(tdExpenses);


                // Column 4: Total Share of Responsibility

                const tdTotal = document.createElement('td');

                tdTotal.dir = cellDir;

                tdTotal.className = cellClass;

                tdTotal.innerHTML = `

                    <span class="val-main"><span class="bidi-num">${hoursFormatted}</span> ${t.hrs}</span>

                    <span class="muted-paren">(<span class="bidi-num">${daysFormatted}</span> ${t.days})</span>

                    <span class="val-main"> + <span class="bidi-num">$${expenseDollarFormatted}</span></span>

                    <span class="word-or">${t.or}</span>

                    <span class="val-main"><span class="bidi-num">$${totalDollarFormatted}</span></span>

                `;

                tr.appendChild(tdTotal);


                tbody.appendChild(tr);

            }

        }

         */

        function updateHitCounters() {

            const counterEl = document.getElementById('stealth-hit-counter');

            if (!counterEl) return;


            try {

                const hitsKey = 'expense_table_hit_count';

                const uniqueCountKey = 'expense_table_unique_device_count';

                const deviceFlagKey = 'expense_table_device_counted';


                const currentHits = (parseInt(localStorage.getItem(hitsKey), 10) || 0) + 1;

                localStorage.setItem(hitsKey, String(currentHits));


                let uniqueDevices = parseInt(localStorage.getItem(uniqueCountKey), 10) || 0;

                if (!localStorage.getItem(deviceFlagKey)) {

                    uniqueDevices += 1;

                    localStorage.setItem(uniqueCountKey, String(uniqueDevices));

                    localStorage.setItem(deviceFlagKey, '1');

                }


                counterEl.textContent = `Hits: ${currentHits} | Unique: ${uniqueDevices}`;

            } catch (err) {

                counterEl.textContent = 'Hits: 1 | Unique: 1';

            }

        }


        // Expose functions explicitly on window for inline handlers

        window.setLanguage = setLanguage;

        window.selectCount = selectCount;


        // Initialize application

        renderButtons();

        renderSubTable();

        renderTable();

        updateHitCounters();

    </script>
